(defun jay/fix-ispell-contraction ()
  "Fix ispell handling of contractions like shouldn't."
  (add-to-list 'ispell-local-dictionary-alist
               '("en_GB" "[[:alpha:]]" "[^[:alpha:]]" "[']" nil ("-d" "en_GB") nil utf-8)))

(defun add-word-to-personal-dictionary ()
  "Add the word at point to the personal dictionary."
  (interactive)
  (require 'ispell)
  (let ((word (or (thing-at-point 'word t)
                  (user-error "No word at point"))))
    ;; Tell the ispell process to add WORD
    (ispell-send-string (format "*%s\n" word))
    ;; Tell ispell to save the personal dictionary now
    (ispell-send-string "#\n")
    (ispell-pdict-save t)
    (message "Added \"%s\" to personal dictionary" word)))

(defun jay/jinx-save-correction-as-abbrev (wrong right)
  "Save WRONG -> RIGHT using Flyspell's abbrev behavior."
  (when (and (require 'flyspell nil t)
             (bound-and-true-p flyspell-abbrev-p))
    (let ((table (flyspell-abbrev-table)))
      (if (flyspell-already-abbrevp table wrong)
          (flyspell-change-abbrev table wrong right)
        (flyspell-define-abbrev wrong right)))))

(defun jay/jinx-correct-word-near-point ()
  "Auto-correct word at or before point using Jinx's first suggestion.
If point is after a space, move back one character first so the previous word
is corrected, matching the old Org writing workflow on the backtick key.
Accepted corrections are also saved as abbrevs, like the old Flyspell path."
  (interactive)
  (when (and (> (point) (point-min))
             (looking-back " " 1))
    (left-char 1))
  (pcase-let ((`(,start . ,end)
               (or (jinx--bounds-of-word)
                   (user-error "No word at point"))))
    (when-let* ((ov (car (ignore-errors
                           (jinx--force-overlays start end))))
                (word (buffer-substring-no-properties
                       (overlay-start ov) (overlay-end ov)))
                (suggestions (jinx--correct-suggestions word)))
      (when-let ((choice (car suggestions)))
        (jay/jinx-save-correction-as-abbrev word choice)
        (jinx--correct-replace ov choice)))))

(defun flyspell-auto-correct-word-correct-space ()
  (interactive)
  (when (looking-back " " nil)
    (left-char 1))
  (flyspell-auto-correct-word))

(defun jay/ensure-jinx-personal-dictionary ()
  "Point Enchant at `ispell-personal-dictionary' so Jinx shares one word list."
  (let* ((personal (expand-file-name
                    (or ispell-personal-dictionary
                        "~/emacs/Spelling/personal.dic")))
         (enchant-dir (expand-file-name "~/.config/enchant"))
         (enchant-dict (expand-file-name "en_US.dic" enchant-dir))
         (ordering-file (expand-file-name "enchant.ordering" enchant-dir))
         (ordering "en:hunspell,aspell\nen_US:hunspell,aspell\n"))
    (when (file-exists-p personal)
      (make-directory enchant-dir t)
      (unless (and (file-exists-p ordering-file)
                   (string=
                    (with-temp-buffer
                      (insert-file-contents ordering-file)
                      (buffer-string))
                    ordering))
        (with-temp-file ordering-file
          (insert ordering)))
      (cond
       ((file-symlink-p enchant-dict)
        (unless (string= (file-truename enchant-dict)
                         (file-truename personal))
          (message "Jinx: keeping existing Enchant dictionary link at %s"
                   enchant-dict)))
       ((file-exists-p enchant-dict)
        (message "Jinx: keeping existing Enchant dictionary at %s"
                 enchant-dict))
       (t
        (make-symbolic-link personal enchant-dict t))))))

(defun my/skip-flyspell-on-large-buffers ()
  "Disable flyspell for large buffers (>5MB) or exported message conversations."
  (when (and flyspell-mode
             (or (> (buffer-size) (* 5 1024 1024))
                 (string-match-p "exported-text-conversations" (or buffer-file-name ""))))
    ;; Remove the hook locally before disabling to avoid re-entering it
    (remove-hook 'flyspell-mode-hook #'my/skip-flyspell-on-large-buffers t)
    (flyspell-mode -1)
    ;; Re-add the hook locally for this buffer
    (add-hook 'flyspell-mode-hook #'my/skip-flyspell-on-large-buffers nil t)))

(add-hook 'flyspell-mode-hook #'my/skip-flyspell-on-large-buffers)
